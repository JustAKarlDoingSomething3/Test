-- ESP Render Loop --

RunService.RenderStepped:Connect(function()
    if not GetValue("ESP") then return end
    
    -- Killers ESP
    if GetValue("KillersESP") and KillersFolder then
        for _, killer in pairs(KillersFolder:GetChildren()) do
            if killer:FindFirstChild("HumanoidRootPart") then
                local color = ColorPresets[GetValue("KillersColor")] or Color3.fromRGB(255, 0, 0)
                CreateDynamicHighlight(true, killer, killer.HumanoidRootPart, {Color = color, MinDistance = 0, MaxDistance = 1000})
                CreateText(true, killer, killer.HumanoidRootPart, {Text = killer.Name, Color = color, MinDistance = 0})
            end
        end
    end

    -- Survivors ESP
    if GetValue("SurvivorsESP") and SurvivorsFolder then
        for _, survivor in pairs(SurvivorsFolder:GetChildren()) do
            if survivor ~= LocalCharacter and survivor:FindFirstChild("HumanoidRootPart") then
                local color = ColorPresets[GetValue("SurvivorsColor")] or Color3.fromRGB(0, 255, 0)
                CreateDynamicHighlight(true, survivor, survivor.HumanoidRootPart, {Color = color, MinDistance = 0, MaxDistance = 1000})
                CreateText(true, survivor, survivor.HumanoidRootPart, {Text = survivor.Name, Color = color, MinDistance = 0})
            end
        end
    end

    -- Generators ESP
    if GetValue("GeneratorsESP") and GameMap then
        local gens = GameMap:FindFirstChild("Generators") or GameMap:FindFirstChild("Gens")
        if gens then
            for _, gen in pairs(gens:GetChildren()) do
                local isCompleted = gen:GetAttribute("Completed") or gen:FindFirstChild("Completed")
                if not (GetValue("GeneratorsCheck") and isCompleted) then
                    local color = ColorPresets[GetValue("GeneratorsColor")] or Color3.fromRGB(0, 255, 255)
                    local root = gen:IsA("Model") and (gen.PrimaryPart or gen:FindFirstChildWhichIsA("BasePart")) or gen
                    if root then
                        CreateDynamicHighlight(true, gen, root, {Color = color, MinDistance = 0, MaxDistance = 1000})
                        CreateText(true, gen, root, {Text = "Generator", Color = color, MinDistance = 0})
